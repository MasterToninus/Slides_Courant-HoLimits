\begin{frame}{From Poisson algebras to higher observables}
  \begin{columns}[T]
    \begin{column}{.46\textwidth}
      \begin{block}{Symplectic geometry}
        A symplectic form $\omega\in\Omega^2_{\mathrm{cl}}(M)$ assigns to
        $f\in C^\infty(M)$ a Hamiltonian vector field
        \[
          \iota_{X_f}\omega=-\d f.
        \]
        The observables form the Poisson algebra
        \[
          \big(C^\infty(M),\{\cdot,\cdot\}_\omega\big).
        \]
      \end{block}
    \end{column}
    \pause
    \begin{column}{.50\textwidth}
      \begin{block}{Higher degree}
        For $\sigma\in\Omega^{r+1}_{\mathrm{cl}}(M)$, Hamiltonian data are pairs
        \[
          (X,\alpha),\qquad \iota_X\sigma=-\d\alpha,
          \qquad \alpha\in\Omega^{r-1}(M).
        \]
        Their brackets assemble into the observable
        $L_\infty$-algebra $\Rogers[\sigma]{r-1}$.
      \end{block}
    \end{column}
  \end{columns}

  \vfill\pause
  \begin{claimblock}
    Higher brackets replace the Poisson bracket and record the geometry of
    observables together with its homotopies.
  \end{claimblock}
\end{frame}
\note[itemize]{
  \item The symplectic case provides the starting point and fixes the sign convention.
  \item For higher degree, functions are replaced by differential forms and the Jacobi identity holds up to coherent homotopies.
}

%-----------------------------------------------------------------------------
\begin{frame}{Higher Courant algebras as ambient structures}
  Fix an integer $r\geq 1$. The relevant standard higher Courant algebroid is
  \[
    E^{(r-1)}=TM\oplus\wedge^{r-1}T^*M.
  \]
  A closed form $\sigma\in\Omega^{r+1}_{\mathrm{cl}}(M)$ twists its bracket and
  determines an $L_\infty$-algebra $\Courant[\sigma]{r-1}$.

  \vfill\pause
  \begin{columns}[T]
    \begin{column}{.47\textwidth}
      \begin{block}{Graphs of closed forms}
        The graph
        \[
          \graph(\sigma)=
          \{X-\iota_X\sigma\mid X\in TM\}
        \]
        is an isotropic involutive subbundle of $E^{(r)}$.
      \end{block}
    \end{column}
    \begin{column}{.49\textwidth}
      \begin{block}{Higher Dirac structures}
        General isotropic involutive subbundles
        \[
          L\subset E^{(n)}
        \]
        extend the graph construction and carry their own observable
        $L_\infty$-algebras.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
\note[itemize]{
  \item Closed forms enter twice: as twisting data for a higher Courant bracket and through their graphs as higher Dirac structures.
  \item The second viewpoint motivates the final part of the talk.
}

%-----------------------------------------------------------------------------
\begin{frame}{Two canonical morphisms}
  The starting point is the sequence
  \[
    \Rogers[\sigma]{r-1}
      \longrightarrow
    \Courant[\sigma]{r-1}
      \longrightarrow
    \Courant{r}.
  \]

  \vfill
  \begin{itemize}
    \item The first morphism embeds multisymplectic observables into the
      corresponding higher Courant algebra \cite{Miti2024}.
    \pause
    \item Zambon asked whether the second morphism exists canonically in every
      degree \cite{Zambon2012}.
    \pause
    \item The preprint \cite{Fiorenza2026} constructs it and explains both
      arrows through homotopy fibers of morphisms between differential graded
      Lie algebras.
  \end{itemize}

  \vfill\pause
  \begin{upshotblock}
    The homotopy-fiber description makes the morphisms natural and suggests a
    path from graphs of closed forms to general higher Dirac structures.
  \end{upshotblock}
\end{frame}
\note[itemize]{
  \item The middle and right algebras have adjacent Courant degrees.
  \item The first arrow is due to Miti--Zambon; the second is the main result of the submitted Fiorenza--Miti preprint.
  \item The higher Dirac extension is ongoing work.
}

\begin{frame}{Plan of the talk}
  \begin{enumerate}
    \item Higher Courant $L_\infty$-algebras and their homotopy-fiber models.
    \vfill
    \item The canonical morphism from the twisted degree-$(r-1)$ algebra to
      the untwisted degree-$r$ algebra.
    \vfill
    \item The multisymplectic observable morphism in the same framework.
    \vfill
    \item Ongoing work on observable algebras of higher Dirac structures.
  \end{enumerate}
\end{frame}
\note[itemize]{
  \item The first three parts concern Fiorenza--Miti, arXiv:2602.14702.
  \item The last part concerns ongoing work with Aditya Basu and Domenico Fiorenza.
  \item Suggested pacing for a 60-minute slot: 12 minutes for background,
    25 for the canonical morphism, 8 for observables, 8 for the higher Dirac
    outlook, and 7 for the conclusion and questions.
}
